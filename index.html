orbLastCandleTime = result.newLastCandleTime || orbLastCandleTime; logger.log('INFO', 'ORB_EVAL_STATUS', result.status + ' — ' + result.message); if (result.status === 'SIGNAL') await processStrategySignal('ORB + VWAP', result); } catch (e) { logger.log('ERROR', 'ENGINE_ERROR', 'ORB+VWAP cycle failed: ' + e.message); } } async function runTrendlineCycle(ltp, candleData) { try { var result = trendline.evaluateTrendlineSignal({ candles: candleData, ltp: ltp, getChainPrice: getChainPrice, lastCandleTime: trendlineLastCandleTime }); trendlineLastCandleTime = result.newLastCandleTime || trendlineLastCandleTime; logger.log('INFO', 'TL_EVAL_STATUS', result.status + ' — ' + result.message); if (result.status === 'SIGNAL') await processStrategySignal('Trendline Breakout', result); } catch (e) { logger.log('ERROR', 'ENGINE_ERROR', 'Trendline cycle failed: ' + e.message); } } async function runQualityBreakoutCycle(ltp, candleData) { try { var todayRes = await trades.getTradesToday('quality_breakout'); } catch (e) { logger.log('ERROR', 'ENGINE_ERROR', 'Quality Breakout cycle failed: ' + e.message); } } async function runCycle() { var bucket = expiry.cycleBucketKey(5); var claimed = await trades.claimCycle('jarvis_nifty_cycle', bucket); IMPORTANT: this only logs right now — there is no real // order-placement code wired up yet. Flipping the toggle currently // changes nothing about actual trading; it's infrastructure for // whenever real execution is built. var statusRes = await db.getEngineStatus(); if (!statusRes.error && statusRes.data && statusRes.data.live_trading_enabled) { logger.log('WARN', 'LIVE_TRADING_WOULD_FIRE_HERE', 'live_trading_enabled=true, but no real order-execution code exists yet — this is a no-op placeholder, not a live trade'); } var loggedIn = await ensureLoggedIn(); if (!loggedIn) { await trades.completeCycle('jarvis_nifty_cycle', bucket, { status: 'FAILED', error: 'Angel One login failed' }); return; } var quote = await marketData.getQuote({ NSE: [tokenMap.TM.NIFTY50.tok] }, 'FULL'); if (!quote || !quote.fetched || !quote.fetched.length) { await trades.completeCycle('jarvis_nifty_cycle', bucket, { status: 'FAILED', error: 'No quote data returned' }); return; } var q = quote.fetched[0]; var ltp = q.ltp; var changePct = q.percentChange != null ?
